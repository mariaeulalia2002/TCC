\begin{simbolos}
  \item[$A$] Área do depósito (m² ou km²)
  \item[$A_s$] Ponto superior adotado na medida da cicatriz
  \item[$B_s$] Ponto basal adotado na medida da cicatriz
  \item[$c'$] Coesão efetiva
  \item[$C(t)$] Cosseno da fase instantânea
  \item[$d_{i,e}$] Distância entre o poço $i$ e o evento $e$
  \item[$f$] Frequência (Hz)
  \item[$f_0$] Frequência fundamental (Hz)
  \item[$\widetilde f_0$] Mediana lognormal da frequência fundamental (Hz)
  \item[$f_{\mathrm{dom}}$] Frequência dominante (Hz)
  \item[$f_{\pm1\sigma}$] Limites de uma unidade de desvio-padrão lognormal (Hz)
  \item[$FS$] Fator de segurança
  \item[$H$] Profundidade estimada da interface (m)
  \item[$H_c$] Ponto médio do envelope de profundidade (m)
  \item[$H_{\min}$] Profundidade mínima do envelope (m)
  \item[$H_{\max}$] Profundidade máxima do envelope (m)
  \item[$H_s$] Altura da cicatriz (m)
  \item[$H_t$] Altura total de queda (m)
  \item[$h$] Espessura convertida para profundidade (m)
  \item[$I$] Integral da espessura temporal sobre a área (ms m²)
  \item[$K_j$] Variância de krigagem no ponto $j$ (ms²)
  \item[$L$] Distância horizontal (m)
  \item[$L_{P-Q}$] Distância horizontal entre os pontos $P$ e $Q$ (m)
  \item[$p_i$] Peso normalizado do poço $i$
  \item[$R_{\mathrm{TWT}}$] Resolução vertical temporal (ms TWT)
  \item[$S$] Inclinação do declive não deformado (°)
  \item[$S_s$] Inclinação máxima da cicatriz (°)
  \item[$S_t$] Inclinação da superfície receptora (°)
  \item[$t$] Tempo duplo de trânsito sísmico (ms TWT)
  \item[$\Delta t$] Diferença de tempo duplo de trânsito (ms TWT)
  \item[$u$] Poropressão
  \item[$u_{\mathrm{pick}}$] Incerteza de repetibilidade da interpretação (ms)
  \item[$u_{\mathrm{interp}}$] Incerteza de interpolação (ms)
  \item[$u_t$] Incerteza temporal (ms)
  \item[$u_v$] Dispersão ponderada da velocidade (m/s)
  \item[$u_z$] Incerteza da profundidade ou espessura convertida (m)
  \item[$u_I$] Incerteza da integral temporal (ms m²)
  \item[$u_V$] Incerteza do volume (km³)
  \item[$u_{\Delta z}$] Incerteza do desnível (m)
  \item[$u_\theta$] Incerteza da inclinação (°)
  \item[$v$] Velocidade sísmica (m/s)
  \item[$\overline v_e$] Velocidade representativa ponderada do evento $e$ (m/s)
  \item[$v_{\mathrm{int}}$] Velocidade intervalar (m/s)
  \item[$V_p$] Velocidade da onda P (m/s)
  \item[$V_s$] Velocidade da onda S (m/s)
  \item[$z$] Profundidade (m)
  \item[$\Delta z$] Diferença de profundidade ou desnível (m)
  \item[$\nu$] Coeficiente de Poisson
  \item[$\sigma_n$] Tensão normal total
  \item[$\sigma'_n$] Tensão normal efetiva
  \item[$\sigma_{\ln(f_0)}$] Desvio-padrão no espaço logarítmico da frequência fundamental
  \item[$\tau_f$] Resistência ao cisalhamento
  \item[$\tau_m$] Tensão cisalhante mobilizada
  \item[$\varphi'$] Ângulo de atrito interno efetivo
  \item[$\phi(t)$] Fase instantânea
  \item[$\Delta H_{\mathrm{abs}}$] Semi-amplitude absoluta do envelope de profundidade (m)
  \item[$\Delta H_{\mathrm{rel}}$] Semi-amplitude relativa do envelope de profundidade (\%)
\end{simbolos}
